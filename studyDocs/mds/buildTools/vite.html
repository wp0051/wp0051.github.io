Vite

是一种新型前端构建工具，能够显著提升前端开发体验

需求

建一个同时支持vue2技术栈、vue3技术栈、react技术栈的Monorepo项目

  • 每一中语言文件夹下modules下的每一个文件夹都是一个完整的项目(当前项目的路径、当前项目的组件、当前项目的vuex或pinia)
  • dev serve只能运行一个文件夹下的入口ts
  • 由于vite对应不同语言有不同的插件，并且不是一个入口文件生成一个vite配置，而是所有入口文件执行一个配置，所以每次build只能做到编译一个项目或者编译一种语言下的所有项目。

支持多项目

  • 项目都放在/packages/目录下
  • 根目录可以按照公共依赖vite、ts、axios等
  • 各个项目可以建立自己的package.json,并且按照自己的依赖vue@2.7+vue-router/vue@^3+vue-router
  • 根目录package.json设置 - npm pnpm cnpm 都会读取使用，但是专注使用pnpm会影响性能

        "packages/*"
    ]

根据环境变量和运行项目判断是否执行

dev必须指定存在的一个项目名npm run dev -- --home

import fg from "fast-glob"   // 递归读取路径规则，返回路径数组，可异步，可同步(fg.sync)
const langs = ['vue2', 'vue3', 'react']; // 当前项目支持的语言

// 封装一个报错函数
function throwError(msg) {
    console.error(`\n🚨 [Vite Config] ${msg}\n`);
    // // 使用非零退出码表示失败
    process.exit(1);
}
export default defineConfig( async({command, mode}) => {
    let argvs = process.argv;
    // 当前项目module下每个文件夹都是一个独立项目，所以启动必须指定moudle，执行npm run dev -- --home
    let last_argv = argvs[argvs.length - 1]; // process.argv返回当前命令的路径或者参数，从最后一个元素获取打包或者编译的moduleName
    let moduleName = last_argv.startsWith("--") ? last_argv.replace(/^--/, "") : ''; // 去掉--
    if (command === 'serve' && mode === 'development' && !moduleName) {
        throwError("错误：开发环境启动服务器必须指定模块启动，例如：npm run dev -- --my");
    }
    const dirs = await checkScript(moduleName);  // 校验执行的命令是否正确
})
// 校验命令执行是否正确
async function checkScript (moduleName){
    // 获取所有入口ts文件，如果执行module，那就全局查找
    let dirReg = ['./packages/*/modules/*/index.ts'];  // 可以是字符串，最好数组，后期不确定一定是ts后缀，所以可能有多个值；
    if(moduleName) dirReg = [`./packages/*/modules/${moduleName}/index.ts`]; // 数组：理由同上；
    const dirs = await fg(dirReg);  // 获取路径列表
    if(!dirs.length){
        throwError("错误：未找到需要打包运行的文件");
    }
    // 根据项目支持语言数组生成语言正则---要求每个子项目文件名必须带语言即可以是packages/vue2也可以是packages/vue2-xxx,不能是packages/v2(逻辑不认)
    let langReg = new RegExp(langs.join("|"));
    if(!dirs.every(v=>langReg.test(v))){
        throwError("错误：需要打包的ts文件路径未声明开发框架");
    }
    // 暂没想到一次build多种语言，暂时只实现打一次
    // 根据支持语言数组，生成校验语言的正则数组
    let langArrReg = langs.map(v=>new RegExp(v));
    // 获取当前所以入口文件使用','拼接的字符串
    let dirstr = dirs.join();
    // 使用reduce获取到入口文件数组包含的语言数量
    // 0为初始值，pre为初始值/上一次返回的值,(pre += (cur.test(dirstr) - 0), pre),','连接的表达式，return最后一个值；
    // (cur.test(dirstr) - 0)  true - 0 = 1;
    if(langArrReg.reduce((pre, cur)=>(pre += (cur.test(dirstr) - 0), pre),0) > 1){
        throwError("错误：一次只能同时打包一种语言");
    }
    return dirs
}

获取build时的多入口文件列表

  • vite入口文件是index.html,支持多入口，但是支持的也是html文件
  • 但是我不希望每个项目创建一个html文件
  • 使用一个文件模板，自动设置script引用
  • 这文件不要写入磁盘，存在内存中

import path from 'path';
import fs from 'fs/promises'; // 仍需读取模板文件
const langs = ['vue2', 'vue3', 'react']; // 当前项目支持的语言

const htmlTemplatePath = path.resolve(__dirname, 'index-template.html');   // build时使用的html模板
const scriptPlaceholder = '<!-- INJECT_SCRIPT -->'    // build时动态插入script标签的占位
const generatedEntries = {}        // build生成虚拟html的列表
export default defineConfig( async({command, mode}) => {
    const entries = await getEntries(dirs); // 获取build时的入口配置
    let plugins = [VirtualHtmlResolver()];  // 自定义组件，用于解析build生成的虚拟html
    // ...其他定制插件、公共插件
    return {
        plugins,  // 插件配置
        base:"./", // 编译完后文件引入的方式-相对路径
        build: {
            rollupOptions: {
                input: entries  // 定义build时多入口-虚拟文件
            }
        },
        resolve: {
            alias: {
                '@packages': path.resolve(__dirname, "packages")   // 为了解决部分系统下路径解读访问不同
            }
        }
    }
})

// 准备生成虚拟html目录列表
async function getEntries(dirs) {
    // 1. 读取并缓存 HTML 模板
    let templateContent = await fs.readFile(htmlTemplatePath, 'utf-8');
    if (!templateContent.includes(scriptPlaceholder)) {
        console.warn(`警告：HTML 模板文件 ${htmlTemplatePath} 中未找到占位符 '${scriptPlaceholder}'。`);
    }
    // 入口文件可以是个 string []，所以这里直接return了；
    return dirs.map(filePath=>{
        const relativeToPackages = path.relative(path.resolve(__dirname, 'packages'), filePath);
        // 使用posix风格（兼容 Unix/Linux 和 Windows）
        const tsPath = path.posix.join('@packages', relativeToPackages);
        const htmlContent = templateContent.replace(
            scriptPlaceholder,
            `<script type="module" src="${tsPath}"></script>` // 使用绝对路径 / 开头更安全
        );
        // path.basename(path[, ext]) 从一个完整路径（无论是文件路径还是目录路径）中提取出最后一部分
        // path.basename(tsPath, path.extname(tsPath)),从tsPath获取最后一部分并且去除后缀名，这里的结果是index
        // path.dirname(tsPath)：获取tsPath文件所在的路径：xxxx/home/index.ts ==> xxxx/home
        const dirName = path.dirname(tsPath);
        // xxxx/home获取最后一部分==>home
        const tsFileName = path.basename(dirName);
        // 我这里没有创建文件，只是把文件内容存在generatedEntries里
        // 怎么实现让vite认识呢？？
        // 使用自定义插件：用时间钩子，在vite读取到我的tsFileName的时候，将htmlContent吐出去 --- 下面的VirtualHtmlResolver插件；
        generatedEntries[tsFileName] = htmlContent;  // 虚拟列表
        return `${tsFileName}.html`;
    })
}



function VirtualHtmlResolver (){
    return {
        name: 'virtual-html-resolver',
        // resolveId 钩子用于解析虚拟模块ID
        resolveId(tsPath){
            if(generatedEntries[tsPath.slice(0, -5)]){
                return tsPath;
            }
            return null
        },
        // load路径文件时，从虚拟列表generatedEntries中读取return；
        load(tsPath) {
            let tsFileName = tsPath.slice(0, -5);
            return generatedEntries[tsFileName] || null;
        }
    }
}
